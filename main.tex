% ========== Section Banner ==========
\newcommand{\ResumeBanner}[2]{%
  \begin{tikzpicture}[baseline=(current bounding box.north)]
    \fill[resbg,rounded corners=1.2mm] (0,0) rectangle (5.15,0.98);
    \fill[#1,rounded corners=0.35mm] (0.07,0.04) rectangle (0.43,0.94);
    \node[text=resarrow,font=\bfseries\fontsize{9}{10}\selectfont,inner sep=0pt]
      at (0.25,0.49) {$>$};
    \node[anchor=west,text=restext,font=\bfseries\fontsize{13}{15}\selectfont,inner sep=0pt]
      at (0.72,0.49) {#2};
    \draw[resline,line width=0.35pt] (0.48,0.04)--(5.08,0.04);
  \end{tikzpicture}%
}

\newcommand{\ResumeSection}[2]{%
  \par\vspace{1.0mm}
  \noindent\ResumeBanner{#1}{#2}
  \par\vspace{0.55mm}%
}

% ========== Basic Information ==========
\newcommand{\ResumeBasicBlock}[2]{%
  \noindent
  \begin{minipage}[t]{0.79\linewidth}
    \vspace{0pt}
    \ResumeBanner{secinfo}{基本信息}
    \par\vspace{2mm}
    #1%
  \end{minipage}%
  \hfill
  \begin{minipage}[t]{0.19\linewidth}
    \vspace{0pt}
    \raggedleft
    #2%
  \end{minipage}
  \par
}

% ========== Education ==========
% #1 Institution  #2 Major / degree / details  #3 Dates
\newcommand{\Edu}[3]{%
  \noindent
  \begin{tabularx}{\linewidth}{@{}
    >{\bfseries\raggedright\arraybackslash}p{0.19\linewidth}
    >{\raggedright\arraybackslash}X
    >{\bfseries\raggedleft\arraybackslash}p{0.22\linewidth}@{}}
    #1 & \hspace{2mm}#2 & #3
  \end{tabularx}%
  \par\vspace{1.0mm}%
}

% ========== Project Experience ==========
% #1 Dates  #2 Project name  #3 Description
\newcommand{\Experience}[3]{%
  \noindent
  \begin{tabularx}{\linewidth}{@{}
    >{\bfseries\raggedright\arraybackslash}p{0.29\linewidth}
    >{\bfseries\raggedright\arraybackslash}X@{}}
    #1 & #2
  \end{tabularx}%
  \par\vspace{0.15mm}
  #3
  \par\vspace{0.4mm}%
}

% ========== Work / Campus Experience ==========
% #1 Dates  #2 Organization  #3 Position  #4 Description
\newcommand{\WorkExperience}[4]{%
  \noindent
  \begin{tabularx}{\linewidth}{@{}
    >{\bfseries\raggedright\arraybackslash}p{0.40\linewidth}
    >{\bfseries\centering\arraybackslash}p{0.30\linewidth}
    >{\bfseries\raggedleft\arraybackslash}X@{}}
    #2 & #3 & #1
  \end{tabularx}%
  \par\vspace{0.2mm}
  #4
  \par\vspace{0.6mm}%
}
